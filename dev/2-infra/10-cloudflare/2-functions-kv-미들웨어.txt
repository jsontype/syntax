1. 규약 (빌드·SDK 없이 그대로 배포)
  apps/<앱>/functions/              ← Pages 가 자동 인식 (Root directory 기준)
    _middleware.js                   모든 요청을 가로채는 미들웨어 (onRequest · next())
    api/license/verify.js            → POST /api/license/verify  (파일 경로 = URL)
    api/shot/[id].js                 → /api/shot/:id             (동적 세그먼트)
    api/alerts/[[path]].js           → /api/alerts/*             (catch-all)
    _lib/*.js                        공용 모듈 (ESM import)
  _routes.json (앱 루트): Functions 로 보낼 경로만 지정 → 정적 파일 요청이 Functions 호출 수(무료 10만/일)를 안 먹는다
    { "version": 1, "include": ["/api/*"], "exclude": [] }
  핸들러 형태
    export async function onRequest({ request, env, next }) { … }     // 모든 메서드
    export async function onRequestPost({ request, env }) { … }       // POST 만
  env 에 바인딩(KV)·변수·시크릿이 들어온다. env.ASSETS.fetch(req) 로 자기 정적 자산을 내부 호출할 수 있다.
  Web 표준만 쓴다: fetch · crypto.subtle · HTMLRewriter · Request/Response · FormData. npm SDK 를 넣지 않는다 (무빌드가 깨진다).

1. 미들웨어 예 ① *.pages.dev 색인 금지 (응답 헤더 추가)
    export async function onRequest(context) {
      const res = await context.next();
      if (new URL(context.request.url).hostname.endsWith('.pages.dev')) {
        const r = new Response(res.body, res);            // 응답은 불변 → 복사해서 헤더를 얹는다
        r.headers.set('X-Robots-Tag', 'noindex, nofollow');
        return r;
      }
      return res;
    }

1. 미들웨어 예 ② OG 메타 현지화 — HTMLRewriter
  카톡/페북 크롤러는 JS 를 안 돌리고 HTML 만 읽는다 → ?lang= 에 따라 응답 직전에 OG 메타를 엣지에서 갈아끼운다.
    class AttrSetter { constructor(v) { this.v = v } element(el) { el.setAttribute('content', this.v) } }
    class TextSetter { constructor(v) { this.v = v } element(el) { el.setInnerContent(this.v) } }
    const M = { en: { t: '…title…', d: '…description…', l: 'en_US', img: 'https://btc.broodev.com/og-en.png' }, ja: { … } };
    async function ogRewrite({ request, next }) {
      const res = await next();
      if (!(res.headers.get('content-type') || '').includes('text/html')) return res;
      const lang = new URL(request.url).searchParams.get('lang'); const m = lang && M[lang];
      if (!m) return res;                                  // 기본 언어는 원본 그대로
      return new HTMLRewriter()
        .on('html', { element(el) { el.setAttribute('lang', lang) } })
        .on('title', new TextSetter(m.t))
        .on('meta[name="description"]', new AttrSetter(m.d))
        .on('meta[property="og:title"]', new AttrSetter(m.t))
        .on('meta[property="og:description"]', new AttrSetter(m.d))
        .on('meta[property="og:locale"]', new AttrSetter(m.l))
        .on('meta[property="og:image"]', new AttrSetter(m.img))
        .transform(res);
    }
  코인 서브도메인은 같은 미들웨어에서 호스트명(eth.broodev.com → eth)으로 canonical/og:url 을 자기 호스트로 바꿔 중복 색인을 막는다.
  실패하면 try/catch 로 원본 응답을 돌려준다 — 미들웨어가 페이지를 죽이면 안 된다.

1. 미들웨어 예 ③ 루트에 띄울 사이트 스위치 (내부 재작성 · URL 그대로)
    const ACTIVE = 'dev3', SITES = ['dev1', 'dev2', 'dev3']
    export async function onRequest({ request, env, next }) {
      const url = new URL(request.url)
      const active = SITES.includes(env && env.DEV_ACTIVE) ? env.DEV_ACTIVE : ACTIVE   // 환경변수가 상수보다 우선 → 커밋 없이 전환
      if (/^\/(dev\d+)(\/|$)/.test(url.pathname)) return withHeaders(await next(), { 'X-Robots-Tag': 'noindex, nofollow' })   // 미리보기
      url.pathname = '/' + active + url.pathname
      const res = await env.ASSETS.fetch(new Request(url.toString(), request))
      const loc = res.headers.get('location')               // 자산 서버가 /dev3/sub → /dev3/sub/ 리다이렉트를 주면 접두어를 벗긴다
      if (loc && res.status >= 300 && res.status < 400) return withHeaders(res, { location: loc.replace(new RegExp('^(https?://[^/]+)?/' + active + '(?=/|$)'), '$1') || '/' })
      if (res.status === 404) {                             // 없는 경로 → 활성 사이트의 404.html (상태 404 유지)
        const nf = await env.ASSETS.fetch(new Request(new URL('/' + active + '/404.html', url).toString()))
        if (nf.ok) return new Response(nf.body, { status: 404, headers: nf.headers })
      }
      return res
    }
    function withHeaders(res, headers) { const r = new Response(res.body, res); for (const k in headers) r.headers.set(k, headers[k]); return r }
  전환: 상수 수정 → 푸시, 또는 Settings → Variables and Secrets → DEV_ACTIVE=dev2 + Retry deployment

1. KV 바인딩 (라이선스 저장소 · 업로드 저장소)
  대시보드 → Storage & Databases → KV → Create (예: utils-licenses)
  Pages 프로젝트 → Settings → Bindings → Add → KV namespace: Variable name UTILS_LICENSES · KV namespace utils-licenses · Production 과 Preview 둘 다
  코드: env.UTILS_LICENSES.get(k) / put(k, value, { expirationTtl: 초 }) / delete(k). 값은 문자열(JSON.stringify) 또는 ArrayBuffer(+ metadata)
  바인딩이 없으면 핸들러가 바로 500 { error: 'kv-not-bound' } 를 돌려주게 해 설정 누락을 드러낸다
  ★ 바인딩·변수는 재배포 뒤에 적용 → Deployments → 최신 배포 → Retry deployment
  키 설계 요령: 본체 lic:<KEY> + 역인덱스 sub:/pi:/cus:/sess:/email: → KEY · 멱등 evt:<id>(TTL) · 레이트리밋 rl:<kind>:<값>(TTL) · 파일 shot:<id>(TTL 90일)
  KV 는 최종 일관성(수 초 지연) · 동시 쓰기 경쟁이 있는 카운터 같은 값에는 쓰지 않는다. 같은 기기 last-seen 은 6시간에 한 번만 기록하는 식으로 쓰기 횟수를 아낀다

1. Turnstile (봇 방지) — 업로드 API 보호 예
  대시보드 → Turnstile → 사이트 추가 → Site key(프론트 위젯) · Secret key(Pages Secret TURNSTILE_SECRET)
  서버: 헤더로 받은 토큰을 siteverify 에 POST → success 아니면 403
    if (env.TURNSTILE_SECRET) {
      const token = request.headers.get('x-turnstile-token') || ''; if (!token) return json({ error: 'turnstile-required' }, 403);
      const form = new FormData(); form.append('secret', env.TURNSTILE_SECRET); form.append('response', token);
      const ip = request.headers.get('cf-connecting-ip'); if (ip) form.append('remoteip', ip);
      const v = await fetch('https://challenges.cloudflare.com/turnstile/v0/siteverify', { method: 'POST', body: form }).then(r => r.json()).catch(() => null);
      if (!v || !v.success) return json({ error: 'turnstile-failed' }, 403);
    }
  시크릿이 미설정이면 강제하지 않고 Origin 검사만 → 위젯 준비 전 배포로 폼이 깨지지 않는다
  Origin/Referer 검사는 위조 가능 → 「다른 사이트가 폼으로 때리는 것」만 막는 최소 방어. 실질 방어선은 Turnstile(업로드) · 서명 검증(웹훅) · 토큰(관리자)

1. 공통 HTTP 유틸 (직접 작성 · 50줄)
    export function json(obj, status, extra) { return new Response(JSON.stringify(obj), { status: status || 200, headers: { 'content-type': 'application/json; charset=utf-8', 'cache-control': 'no-store', ...(extra || {}) } }); }
    export function err(code, status, extra) { return json({ ok: false, error: code, ...(extra || {}) }, status || 400); }
    export async function readJson(request, maxBytes) {                      // 크기 제한(기본 16KB) · 잘못된 JSON → null
      const limit = maxBytes || 16 * 1024; if (Number(request.headers.get('content-length') || 0) > limit) return null;
      const text = await request.text(); if (text.length > limit) return null;
      try { return JSON.parse(text || '{}'); } catch (e) { return null; } }
    export function sameSite(request) {                                        // Origin/Referer 가 자기 도메인·*.pages.dev·localhost 인지 (없으면 통과)
      const ref = request.headers.get('origin') || request.headers.get('referer') || ''; if (!ref) return true;
      return /^https?:\/\/([a-z0-9-]+\.)*broodev\.com(\/|$)|^https:\/\/[a-z0-9.-]+\.pages\.dev(\/|$)|^https?:\/\/(localhost|127\.0\.0\.1)(:\d+)?(\/|$)/i.test(ref); }
    export function timingSafeEqual(a, b) {                                    // 토큰·서명 비교는 상수 시간
      a = String(a); b = String(b); const len = Math.max(a.length, b.length); let diff = a.length === b.length ? 0 : 1;
      for (let i = 0; i < len; i++) diff |= (a.charCodeAt(i) || 0) ^ (b.charCodeAt(i) || 0); return diff === 0; }
    export function maskEmail(email) { const s = String(email || ''); const at = s.indexOf('@'); if (at <= 0) return s ? '***' : ''; return s.slice(0, Math.min(2, at)) + '***' + s.slice(at); }
  같은 도메인에서만 호출하므로 CORS 헤더는 두지 않는다. 응답은 전부 no-store.

1. 로컬 테스트
  Functions 는 ESM + Web 표준만 쓰므로 Node 22 에서 핸들러 함수를 직접 import 하고 { request: new Request(url, {…}), env: { KV: 가짜(Map), 시크릿… } } 를 넘겨 돌린다
  가짜 KV: get/put/delete 를 Map 으로, expirationTtl 은 무시. 서명이 필요한 웹훅은 서명 생성 함수(signPayload)로 헤더를 만든다
  wrangler pages dev 도 되지만 레포에 wrangler 설정을 두지 않는 방침이라 쓰지 않았다
